rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // 기존: 앱이 기기ID로 코드 저장
    match /rewards/{deviceId} {
      allow get: if true;
      allow create: if true;
    }

    // 신규: 앱이 코드를 키로 저장 (웹에서 코드 검증용)
    match /code_index/{code} {
      allow get: if true;
      allow create: if true;
    }

    // 신규: 웹에서 주소 제출 저장 (중복 확인 + 신청)
    match /claims/{code} {
      allow get: if true;
      allow create: if true;
    }

  }
}
